    return crypto.subtle.importKey('jwk', jwk,
      { name: 'RSA-OAEP', hash: 'SHA-256' }, false, ['encrypt']);

    return crypto.subtle.importKey('jwk', jwk,
      { name: 'RSA-OAEP', hash: 'SHA-256' }, false, ['decrypt']);

  async function e2eEncrypt(plaintext, recipientPubJwk) {
    const aesKey = await crypto.subtle.generateKey(
      { name: 'AES-GCM', length: 256 }, true, ['encrypt', 'decrypt']);
    const iv = crypto.getRandomValues(new Uint8Array(12));
    const encoded = new TextEncoder().encode(plaintext);
    const cipher = await crypto.subtle.encrypt({ name: 'AES-GCM', iv }, aesKey, encoded);

    const b64 = (buf) => btoa(String.fromCharCode(...new Uint8Array(buf)));
    return { c: b64(cipher), k: b64(encAes), i: b64(iv) };

  let audioCtx = null;
  function beep(freq, dur) {
    try {
      if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.type = 'sine';
      osc.frequency.value = freq || 880;
      gain.gain.setValueAtTime(0.0001, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.08, audioCtx.currentTime + 0.01);
      gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + (dur || 0.15));
      osc.start();
      osc.stop(audioCtx.currentTime + (dur || 0.15));
    } catch(e) {}
  }

  function fmtTime(t) {
    if (typeof t === 'number') {

  function createTab(opts) {
    const id = (opts.type === 'channel') ? 'channel' : ('pm:' + opts.with.toLowerCase());
    if (tabs.find(t => t.id === id)) return tabs.find(t => t.id === id);

    const tabBtn = el('div', 'tab' + (opts.type === 'pm' ? ' pm' : ''));
    tabBtn.dataset.id = id;
    if (opts.type === 'pm') tabBtn.append(el('span', 'tab-icon', '💬'));
    tabBtn.append(el('span', 'tab-name', opts.name));
    if (opts.type === 'pm') {
      const closeBtn = el('span', 'tab-close', '×');
      closeBtn.addEventListener('click', (e) => { e.stopPropagation(); closeTab(id); });
      tabBtn.append(closeBtn);
    }
    tabBtn.addEventListener('click', () => activateTab(id));

    const pane = el('div', 'pane');
    pane.dataset.id = id;
    const log = el('div', 'log');
    const jump = el('button', 'jump', '↓ new messages');
    jump.hidden = true;
    jump.addEventListener('click', () => { log.scrollTop = log.scrollHeight; jump.hidden = true; });
    log.addEventListener('scroll', () => { if (nearBottom(log)) jump.hidden = true; });

    pane.append(log, jump);
    mainEl.append(pane);
    tabsEl.append(tabBtn);

    const t = { id, type: opts.type, with: opts.with || null, name: opts.name, unread: 0, el: tabBtn, pane, log, jump };
    tabs.push(t);
    if (opts.type === 'pm') sendWS({ type: 'pm_history', with: opts.with });
    activateTab(id);
    return t;
  }

  function activateTab(id) {
    activeTabId = id;
    tabs.forEach(t => {
      const active = (t.id === id);
      t.el.classList.toggle('active', active);
      t.pane.classList.toggle('active', active);
      if (active) {
        t.unread = 0;
        updateUnreadBadge(t);
        updateInputMode(t);
        t.log.scrollTop = t.log.scrollHeight;
        t.jump.hidden = true;
      }
    });
    recalcTotalUnread();
    input.focus();
  }

  function closeTab(id) {
    const idx = tabs.findIndex(t => t.id === id);
    if (idx === -1) return;
    if (tabs[idx].type === 'channel') return;
    tabs[idx].el.remove();
    tabs[idx].pane.remove();
    tabs.splice(idx, 1);
    if (activeTabId === id) {
      const channel = tabs.find(t => t.type === 'channel');
      if (channel) activateTab(channel.id);
    }
  }

  function updateUnreadBadge(t) {
    const old = t.el.querySelector('.unread');
    if (old) old.remove();
    if (t.unread > 0 && activeTabId !== t.id) t.el.append(el('span', 'unread'));
  }

  }

  /* ===== WEBSOCKET ===== */
  let ws = null;
  let reconnectAttempts = 0;
  let reconnectTimer = null;
  let lastConnectTime = 0;
  const pendingQueue = [];

  function sendWS(obj) {
    if (ws && ws.readyState === 1) ws.send(JSON.stringify(obj));
    else pendingQueue.push(obj);
  }

  function connect() {
    if (ws) {
      try {
        ws.onopen = null; ws.onmessage = null;
        ws.onerror = null; ws.onclose = null;
        if (ws.readyState === 0 || ws.readyState === 1) ws.close();
      } catch(e) {}
      ws = null;
    }
    if (reconnectTimer) { clearTimeout(reconnectTimer); reconnectTimer = null; }

    setStatus('', 'connecting');
    ws = new WebSocket(WS_URL);

    ws.onopen = () => {
      reconnectAttempts = 0;
      lastConnectTime = Date.now();
      setStatus('on', 'connected');
      ws.send(JSON.stringify({ type:'auth', username:NICK, password:TOKEN }));
      while (pendingQueue.length) ws.send(JSON.stringify(pendingQueue.shift()));
    };

    ws.onmessage = e => {
      let m;
      try { m = JSON.parse(e.data); } catch (err) { return; }

      if (m.type === 'auth_fail') {
        setStatus('off', 'auth failed');
        const t = getTab('channel');
        if (t) renderSysInto(t, nowHHMM(), 'part', 'system', 'Auth failed: ' + (m.reason || 'unknown'));
        return;
      }
      if (m.type === 'welcome') {
        onlineUsers.add(m.nick);
        setUsers([...onlineUsers]);
        const t = getTab('channel');
        if (t) renderSysInto(t, nowHHMM(), 'join', m.nick, 'has joined ' + CHANNEL);
        ensureKeyPair().then(({ pub }) => {
          sendWS({ type:'pubkey', pubkey: JSON.stringify(pub) });
        });
        return;
      }
      if (m.type === 'users') {
        onlineUsers.clear();
        m.users.forEach(u => onlineUsers.add(u));
        setUsers([...onlineUsers]);
        return;
      }
      if (m.type === 'pubkeys') {
        for (const [user, key] of Object.entries(m.keys)) {
          if (key) { try { pubkeyCache.set(user.toLowerCase(), JSON.parse(key)); } catch(e) {} }
        }
        return;
      }
      if (m.type === 'msg') {
        const t = getTab('channel');
        if (!t) return;
        renderMsgInto(t, m.time, m.nick, m.text, m.nick === NICK, m.color, m.avatar);
        if (m.nick !== NICK) {
          if (activeTabId === 'channel') pulse();
          if (document.hidden || activeTabId !== 'channel') {
            beep(660, 0.12);
            t.unread++; updateUnreadBadge(t); recalcTotalUnread();
          }
        }
        return;
      }
      if (m.type === 'pm') {
        const other = (m.nick === NICK) ? m.target : m.nick;
        let t = getTab('pm:' + other.toLowerCase());
        if (!t) t = createTab({ type:'pm', with:other, name:other });

        const render = (text) => {
          renderMsgInto(t, m.time, m.nick, text, m.nick === NICK, m.color, m.avatar);
        };

        if (m.enc && m.nick !== NICK) {
          e2eDecrypt(m.enc).then(render).catch(() => render('[decrypt failed]'));
        } else if (m.enc && m.nick === NICK) {
          e2eDecrypt(m.enc).then(render).catch(() => render('[encrypted]'));
        } else {
          render(m.text);
        }

        if (m.nick !== NICK) {
          if (activeTabId !== t.id) {
            beep(880, 0.15);
            setTimeout(() => beep(1100, 0.15), 160);
            t.unread++; updateUnreadBadge(t); recalcTotalUnread();
            notify('PM from ' + m.nick, '[encrypted]');
          } else {
            pulse();
          }
        }
        return;
      }
      if (m.type === 'pm_history') {
        const t = getTab('pm:' + m.with.toLowerCase());
        if (!t) return;
        m.msgs.forEach(x => {
          if (x.enc) {
            e2eDecrypt(x.enc).then(txt => {
              renderMsgInto(t, x.time, x.nick, txt, x.nick === NICK, x.color, x.avatar);
            }).catch(() => {
              renderMsgInto(t, x.time, x.nick, '[encrypted]', x.nick === NICK, x.color, x.avatar);
            });
          } else {
            renderMsgInto(t, x.time, x.nick, x.text, x.nick === NICK, x.color, x.avatar);
          }
        });
        return;
      }
      if (m.type === 'pm_fail') {
        const t = getActiveTab();
        if (t) renderSysInto(t, nowHHMM(), 'part', 'system', m.reason);
        return;
      }
      if (m.type === 'typing') {
        if (m.nick === NICK) return;
        const key = m.nick.toLowerCase();
        if (m.state === 'start') typingUsers.set(key, Date.now());
        else typingUsers.delete(key);
        showTyping();
        clearTimeout(typingDisplayTimer);
        typingDisplayTimer = setTimeout(() => {
          for (const [k, ts] of typingUsers) {
            if (Date.now() - ts > 3000) typingUsers.delete(k);
          }
          showTyping();
        }, 3000);
        return;
      }
      if (m.type === 'error') {
        const t = getActiveTab();
        if (t) renderSysInto(t, nowHHMM(), 'part', 'system', m.reason || 'Error');
        return;
      }
      if (m.type === 'sys') {
        if (m.tag === 'join') onlineUsers.add(m.nick);
        if (m.tag === 'part') onlineUsers.delete(m.nick);
        setUsers([...onlineUsers]);
        const t = getTab('channel');
        if (t) {
          renderSysInto(t, m.time, m.tag, m.nick, m.text);
          if (activeTabId === 'channel') pulse();
        }
        return;
      }
    };

    ws.onclose = (event) => {
      console.warn('WS closed. Code:', event.code, 'Reason:', event.reason || '(none)');
      setStatus('off', 'disconnected, retrying');
      reconnectAttempts++;
      const elapsed = Date.now() - lastConnectTime;
      const minDelay = Math.max(0, 3000 - elapsed);
      const backoff = Math.min(1500 * Math.pow(1.5, reconnectAttempts - 1), 10000);
      reconnectTimer = setTimeout(connect, Math.max(minDelay, backoff));
    };
    ws.onerror = (e) => { console.error('WS error:', e); setStatus('off', 'error'); };
  }

  async function refreshWsToken() {
    try {
      const fd = new FormData();
      fd.append('csrf_token', CSRF);

      const d = await r.json();
      if (d.ok && d.ws_token) {
        TOKEN = d.ws_token;
        console.log('[ws] token refreshed, expires in', d.expires_in, 's');
      }
    } catch (e) {
      console.warn('[ws] token refresh failed', e);
    }
  }
  setInterval(refreshWsToken, 20 * 60 * 1000);

  connect();

  let sending = false;
  let lastTypingSent = 0;
  const TYPING_THROTTLE = 2000;

  input.addEventListener('input', () => {
    if (!input.value.trim()) return;
    const now = Date.now();
    if (now - lastTypingSent > TYPING_THROTTLE) {
      lastTypingSent = now;
      sendWS({ type:'typing', state:'start' });
    }
    clearTimeout(typingTimer);
    typingTimer = setTimeout(() => { sendWS({ type:'typing', state:'stop' }); }, 2500);
  });

  input.addEventListener('keydown', (e) => {
    if (e.key === 'Tab') {
      e.preventDefault();
      const val = input.value;
      const match = val.match(/@([a-zA-Z0-9_]*)$/);
      if (!match) return;
      const partial = match[1].toLowerCase();
      if (!partial) return;
      const found = [...onlineUsers].find(u => u.toLowerCase().startsWith(partial) && u.toLowerCase() !== NICK.toLowerCase());
      if (found) {
        input.value = val.slice(0, val.length - match[1].length) + found + ' ';
      }
    }
  });

  document.addEventListener('keydown', (e) => {
    if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
      e.preventDefault();
      openCmdModal('');
      return;
    }
    if (e.key === 'Escape' && cmdBg.classList.contains('open')) closeCmdModal();
  });

  form.addEventListener('submit', e => {
    e.preventDefault();
    if (sending) return;
    const v = input.value.trim();
    if (!v) return;
    const t = getActiveTab();
    if (!t) return;

    if (v.charAt(0) === '/') {
      input.value = '';
      const cmd = v.split(' ')[0].toLowerCase();
      if (cmd === '/help' || cmd === '/?' || cmd === '/commands') { openCmdModal(''); return; }
      if (cmd === '/clear') { confirmClear(); return; }
      if (cmd === '/logout' || cmd === '/exit' || cmd === '/quit') { confirmLogout(); return; }
      if (cmd === '/who' || cmd === '/users' || cmd === '/online') { openWhoModal(); return; }
      if (cmd === '/profile' || cmd === '/me' || cmd === '/nick') {
        openProfile(NICK); setTimeout(() => showEditForm(), 100); return;
      }
      if (cmd === '/pm') {
        const target = v.split(' ')[1];
        if (target) {
          const name = target.replace(/^@/, '');
          if (name) { openPmTab(name); return; }
        }
        toast('Usage: /pm @username', 'err');
        return;
      }
      openCmdModal(v.slice(1));
      return;
    }

    sending = true;
    input.value = '';
    sendWS({ type:'typing', state:'stop' });

    if (t.type === 'pm') {
      const targetKey = pubkeyCache.get(t.with.toLowerCase());
      if (targetKey) {
        e2eEncrypt(v, targetKey).then(enc => {
          sendWS({ type:'pm', target: t.with, enc });
        }).catch(() => {
          sendWS({ type:'pm', target: t.with, text: v });
        });
      } else {
        sendWS({ type:'pm', target: t.with, text: v });
        requestPubkeys([t.with]);
      }
    } else {
      sendWS({ type:'msg', text: v });
    }

    setTimeout(() => { sending = false; }, 200);
  });
})();
